KEYS = ["id", "profile", "pid", "creationFiletime", "hwndHex", "ownershipStartFiletime", "issuedFiletime", "expiresFiletime", "durationSeconds", "purpose"]
HEADER = "AssetLibrary.Explorer.DebugPlan.v1"


def validate(data):
    if len(data) > 2048 or data.startswith(b"\xef\xbb\xbf"):
        raise ValueError("Plan size or BOM refused")
    lines = data.decode("utf-8").splitlines()
    if len(lines) != 11 or lines[0] != HEADER or any(not line for line in lines):
        raise ValueError("Exactly 11 nonempty lines required")
    fields = {}
    for key, line in zip(KEYS, lines[1:]):
        name, separator, value = line.partition("=")
        if name != key or separator != "=" or not value or "=" in value:
            raise ValueError("Ordered single-value field required")
        fields[key] = value
    if not re.fullmatch(r"[0-9a-f]{32}", fields["id"]):
        raise ValueError("Invalid plan id")
    for key in ["pid", "creationFiletime", "ownershipStartFiletime", "issuedFiletime", "expiresFiletime"]:
        if not re.fullmatch(r"[1-9][0-9]*", fields[key]) or int(fields[key]) > 2**63 - 1:
            raise ValueError("Invalid positive decimal field: " + key)
    if int(fields["pid"]) > 2**32 - 1 or not re.fullmatch(r"[1-9A-F][0-9A-F]{0,15}", fields["hwndHex"]):
        raise ValueError("Invalid PID or HWND")
    own, created, issued, expires = [int(fields[k]) for k in ["ownershipStartFiletime", "creationFiletime", "issuedFiletime", "expiresFiletime"]]
    if not own <= created <= issued or issued - own > 6000000000 or expires - issued != 1200000000:
        raise ValueError("Invalid ownership or plan time limits")
    if fields["profile"] != "official" or fields["durationSeconds"] != "60" or fields["purpose"] != "owned-explorer-namespace-diagnostic":
        raise ValueError("Frozen plan settings changed")
    return fields


def build(plan_id, identity, issued):
    hwnd = str(identity["NativeHwnd"])
    if not re.fullmatch(r"[1-9][0-9]*", hwnd):
        raise ValueError("Identity HWND must be one positive decimal value")
    lines = [
        HEADER,
        "id=" + plan_id,
        "profile=official",
        "pid=" + str(identity["Pid"]),
        "creationFiletime=" + str(identity["CreationFileTime"]),
        "hwndHex=" + format(int(hwnd), "X"),
        "ownershipStartFiletime=" + str(identity["OwnershipStartFiletime"]),
        "issuedFiletime=" + str(issued),
        "expiresFiletime=" + str(issued + 1200000000),
        "durationSeconds=60",
        "purpose=owned-explorer-namespace-diagnostic",
    ]
    data = ("\n".join(lines) + "\n").encode("utf-8")
    validate(data)
    return data

def main():

    args = parser.parse_args()
