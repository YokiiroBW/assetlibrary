$record=Get-Content -LiteralPath (Join-Path $run ($Label+'.json')) -Raw|ConvertFrom-Json
$process=Get-Process -Id $record.Pid
$handle=$process.SafeHandle
try{
 if($process.StartTime.ToUniversalTime().ToFileTimeUtc().ToString() -cne [string]$record.CreationFileTime -or
 $process.Path -ine 'C:\YOKI\Codex\AssetLibrary-worktrees\V03-005\.runtime\explorer-g2-faults\live-01\binaries\ExplorerFaultHost.exe' -or $process.SessionId -ne 2){throw 'Owned fault identity mismatch'}
