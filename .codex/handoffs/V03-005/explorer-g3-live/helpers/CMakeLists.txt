cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(G3ReadOnlyObservers LANGUAGES CXX)
add_executable(WindowResponse WindowResponse.cpp)
add_executable(ProofG3View ProofG3View.cpp)
foreach(target WindowResponse ProofG3View)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_compile_options(${target} PRIVATE /W4 /WX /permissive- /analyze /utf-8)
endforeach()
target_link_libraries(WindowResponse PRIVATE user32 advapi32 psapi wtsapi32)
target_link_libraries(ProofG3View PRIVATE ole32 oleaut32 shell32 shlwapi uuid)
