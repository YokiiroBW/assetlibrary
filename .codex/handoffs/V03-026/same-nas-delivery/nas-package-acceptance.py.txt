class Api(prior.Api):
 def request(self,path,payload=None,*,origin=ORIGIN,csrf=None):
  return super().request(path,payload,origin=origin,csrf=csrf)

def cli(*args,input=None,timeout=240):
 p=prior.nas(['sh',DEPLOY+'/nasctl.sh',*args],input=input,timeout=timeout)

 print('CLI '+args[0]+' completed',flush=True)

def images(api,library,entries):
 results=[]
 for name in [*checks.SUCCESS,*checks.ERRORS]:
  entry=entries[name]
  for variant,profile in [('thumbnail',0),('preview',1)]:
   path=f'/assetlink/v1/libraries/{library}/entries/{entry}/image?variant={variant}'
   request=urllib.request.Request(ORIGIN+path,headers={'Origin':ORIGIN,'X-AssetLibrary-CSRF':api.csrf})
   try:response=api.opener.open(request,timeout=17)
   except urllib.error.HTTPError as e:response=e
   with response:
    data=response.read(12582913);status=response.status;assert len(data)<=12582912
    row={'file':name,'variant':variant,'status':status,'bytes':len(data)}
    if name in checks.SUCCESS:
     assert status==200,(name,status,data[:100]);assert response.headers.get_content_type()=='image/png'
     checks.validate_png(data,checks.SUCCESS[name][profile],2097152 if profile==0 else 12582912)
     row['sha256']=hashlib.sha256(data).hexdigest();row['dimensions']=checks.SUCCESS[name][profile]
    else:
     expected={'active.svg':415,'not-an-image.png':415,'oversized-header.png':422,'truncated.jpg':422}[name]
     assert status==expected,(name,status)
    results.append(row)
 return results

def verify():
 prior.ORIGIN=ORIGIN
 api=Api(OUT/'acceptance.crt');assert api.request('/assetlink/v1/auth/session')[0]==401;api.login(False)
 try:
  status,reply=api.control('libraries.list',{'page_size':100});assert status==200
  libraries=reply['body']['items']
  if not libraries:
   status,reply=api.control('libraries.register',{'source_key':'fixture','display_name':'Synthetic images','root_path':'/assets/fixture','category':'images'},mutation=True)
   assert status==200,(status,reply);library=reply['body']['library_id']
   status,reply=api.control('library_scans.start',{'library_id':library},mutation=True);assert status==200
  else:
   assert len(libraries)==1;library=libraries[0]['library_id']
  deadline=time.monotonic()+120
  while True:
   status,reply=api.control('library_scans.get',{'library_id':library});assert status==200
   scan=reply['body']['scan']
   if scan and scan['state']=='succeeded':break
   assert scan and scan['state'] in ('queued','leased') and time.monotonic()<deadline,(scan and scan['state'])
   time.sleep(.5)
  pending=[''];entries={}
  while pending:
   parent=pending.pop();status,reply=api.control('entries.browse',{'library_id':library,'parent_relative_path':parent,'page_size':100})
   assert status==200 and reply['body']['next_cursor'] is None
   for entry in reply['body']['items']:
    if entry['kind']=='directory':pending.append(entry['relative_path'])
    else:entries[entry['relative_path']]=entry['entry_id']
  results=images(api,library,entries)
  assert api.request(f'/assetlink/v1/libraries/{library}/entries/{uuid.uuid4()}/image?variant=thumbnail')[0]==404
  assert api.request(f'/assetlink/v1/libraries/{library}/entries/{entries["landscape.png"]}/image?variant=wrong')[0]==400
  (OUT/'fixture-entries.json').write_text(json.dumps({'library':library,'entries':entries},indent=2)+'\n',encoding='utf-8')
  report={'status':'image_cases_passed','origin':ORIGIN,'library_id':library,'cases':results,'scan':{k:scan[k] for k in ('scan_id','state','observed_entries','committed_entries')}}
  (OUT/'api-images.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
  print(json.dumps({'status':report['status'],'cases':len(results),'scan_count':scan['committed_entries']}),flush=True)
 finally:assert api.request('/assetlink/v1/auth/logout',{})[0]==204
 assert api.request('/assetlink/v1/auth/session')[0]==401

 prior.nas(['docker','stop','--time','10',item['Id']])
 api=Api(OUT/'acceptance.crt');api.login(False)
