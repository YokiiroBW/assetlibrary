template<typename T> static T ReadSelf(ULONG_PTR address) {

    Need(ReadProcessMemory(GetCurrentProcess(),reinterpret_cast<LPCVOID>(address),&value,sizeof(value),&copied)&&copied==sizeof(value),"bounded_self_image_read");

}
static std::string Hex(const std::array<BYTE,32>& bytes) {
    static constexpr char digits[]="0123456789abcdef"; std::string result;

}
struct Sha256 {
    BCRYPT_ALG_HANDLE algorithm=nullptr; BCRYPT_HASH_HANDLE hash=nullptr; std::vector<BYTE> state;
    Sha256(){
        Need(BCryptOpenAlgorithmProvider(&algorithm,BCRYPT_SHA256_ALGORITHM,nullptr,0)>=0,"sha256_provider");
        DWORD size=0,received=0;
        Need(BCryptGetProperty(algorithm,BCRYPT_OBJECT_LENGTH,reinterpret_cast<PUCHAR>(&size),sizeof(size),&received,0)>=0&&received==sizeof(size)&&size<65536,"sha256_state_size");
        state.resize(size);
        Need(BCryptCreateHash(algorithm,&hash,state.data(),size,nullptr,0,0)>=0,"sha256_create");
    }
    ~Sha256(){if(hash)BCryptDestroyHash(hash);if(algorithm)BCryptCloseAlgorithmProvider(algorithm,0);}
    void Add(BYTE* bytes,ULONG size){Need(BCryptHashData(hash,bytes,size,0)>=0,"sha256_add");}
    std::string Finish(){std::array<BYTE,32> result{};Need(BCryptFinishHash(hash,result.data(),static_cast<ULONG>(result.size()),0)>=0,"sha256_finish");return Hex(result);}
};
