function Assert-Tree($key,[string]$relative,$nodes) {
  if(-not $nodes.ContainsKey($relative)){throw 'Foreign registry node prevents cleanup'}
  $values=$nodes[$relative]
  foreach($name in $key.GetValueNames()) {
    if(-not $values.ContainsKey($name)){throw 'Foreign registry value prevents cleanup'}
    $expected=$values[$name]
    if($key.GetValue($name) -cne $expected){throw 'Changed registry value prevents cleanup'}
    $kind=if($expected -is [int]){[Microsoft.Win32.RegistryValueKind]::DWord}else{[Microsoft.Win32.RegistryValueKind]::String}
    if($key.GetValueKind($name) -ne $kind){throw 'Changed registry value type prevents cleanup'}
  }
  foreach($childName in $key.GetSubKeyNames()) {
    $child=$key.OpenSubKey($childName,$false)
    $childPath=if($relative){"$relative\$childName"}else{$childName}
    try{Assert-Tree $child $childPath $nodes}finally{$child.Dispose()}
  }
}
