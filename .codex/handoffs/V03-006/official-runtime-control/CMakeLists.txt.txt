cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(OfficialActiveView LANGUAGES CXX)
add_executable(OfficialActiveView OfficialView.cpp)
target_compile_features(OfficialActiveView PRIVATE cxx_std_17)
target_compile_definitions(OfficialActiveView PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OfficialActiveView PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(OfficialActiveView PRIVATE ole32 oleaut32 shell32 shlwapi uuid)
