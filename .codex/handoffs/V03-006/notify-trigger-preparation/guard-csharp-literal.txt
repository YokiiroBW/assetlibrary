 public static RegistryKey CreateExclusive(string path) {
   IntPtr key; int disposition;
   int error=RegCreateKeyEx(new IntPtr(unchecked((int)0x80000001)),path,0,null,0,0x2011F,IntPtr.Zero,out key,out disposition);
   if(error!=0)throw new System.ComponentModel.Win32Exception(error);
   if(disposition!=1){RegCloseKey(key);throw new InvalidOperationException("Registry creation raced with an existing root");}
   return RegistryKey.FromHandle(new SafeRegistryHandle(key,true),RegistryView.Registry64);
 }

 public static NotificationReport NotifyDrives() {
  var result=new NotificationReport();var watch=System.Diagnostics.Stopwatch.StartNew();
  var thread=new Thread(()=>{
   IntPtr pidl=IntPtr.Zero;bool initialized=false;
   try{
    lock(result){result.Apartment=Thread.CurrentThread.GetApartmentState().ToString();result.ManagedThreadId=Thread.CurrentThread.ManagedThreadId;result.NativeThreadId=GetCurrentThreadId();int type,qualifier;result.ApartmentHr=CoGetApartmentType(out type,out qualifier);result.ApartmentType=type;result.Stage="apartment";}
    int init=CoInitializeEx(IntPtr.Zero,2);lock(result){result.CoInitializeHr=init;result.Stage="CoInitializeEx-returned";}if(init<0)return;initialized=true;
    int hr=SHGetSpecialFolderLocation(IntPtr.Zero,17,out pidl);lock(result){result.FolderHr=hr;result.PidlPresent=pidl!=IntPtr.Zero;result.Stage="GetSpecialFolderLocation-returned";}if(hr<0||pidl==IntPtr.Zero)return;

    lock(result){result.AssocNotifyReturned=true;result.AssocReturnedElapsedMs=watch.ElapsedMilliseconds;result.Stage="SHCNE_ASSOCCHANGED-returned";}
   }catch(Exception error){lock(result){result.Stage="exception";result.ExceptionType=error.GetType().FullName;}}
   finally{if(pidl!=IntPtr.Zero)Marshal.FreeCoTaskMem(pidl);if(initialized)CoUninitialize();}
  });

  lock(result){result.Joined=joined;result.ElapsedMs=watch.ElapsedMilliseconds;return result.Copy();}
 }
