cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(OfficialAttributePidlControl LANGUAGES CXX)
add_executable(OfficialAttributePidlControl AttributePidlControl.cpp)
set_property(TARGET OfficialAttributePidlControl PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_features(OfficialAttributePidlControl PRIVATE cxx_std_17)
target_compile_definitions(OfficialAttributePidlControl PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OfficialAttributePidlControl PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(OfficialAttributePidlControl PRIVATE ole32 shell32 uuid bcrypt)
