cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(OfficialParentControl LANGUAGES CXX)
add_executable(OfficialParentControl ParentControl.cpp)
target_compile_features(OfficialParentControl PRIVATE cxx_std_17)
target_compile_definitions(OfficialParentControl PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OfficialParentControl PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(OfficialParentControl PRIVATE ole32 shell32 shlwapi uuid)

add_executable(OfficialParentVisible ParentVisible.cpp)
target_compile_features(OfficialParentVisible PRIVATE cxx_std_17)
target_compile_definitions(OfficialParentVisible PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OfficialParentVisible PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(OfficialParentVisible PRIVATE ole32 shell32 shlwapi uuid)
