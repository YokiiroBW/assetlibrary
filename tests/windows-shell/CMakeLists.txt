# Read the coordinator's independent byte vectors, never produce fixtures from the codec.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../contracts/windows-shell/thumbnail-vectors-v1.json" thumbnail_vectors)
string(JSON vector_count LENGTH "${thumbnail_vectors}" vectors)
math(EXPR vector_last "${vector_count}-1")
set(vector_header "#pragma once\nstruct ThumbnailVector {const char* name; const char* direction; bool valid; const char* hex;};\ninline constexpr ThumbnailVector ThumbnailVectors[] = {\n")
foreach(index RANGE 0 ${vector_last})
  foreach(field name direction hex valid)
    string(JSON vector_${field} GET "${thumbnail_vectors}" vectors ${index} ${field})
  endforeach()
  if(vector_valid)
    set(vector_valid_cpp true)
  else()
    set(vector_valid_cpp false)
  endif()
  string(APPEND vector_header "{\"${vector_name}\",\"${vector_direction}\",${vector_valid_cpp},\"${vector_hex}\"},\n")
endforeach()
string(APPEND vector_header "};\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ThumbnailVectors.generated.h" CONTENT "${vector_header}")
