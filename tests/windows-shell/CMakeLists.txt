cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(AssetLibraryExplorerProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ExplorerSnapshot STATIC Snapshot.cpp SnapshotPidl.cpp SnapshotPipe.cpp)
target_compile_definitions(ExplorerSnapshot PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExplorerSnapshot PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerSnapshot PUBLIC ole32 advapi32 uuid)
add_library(ExplorerLoadingRefresh STATIC LoadingRefresh.cpp)
target_compile_options(ExplorerLoadingRefresh PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerLoadingRefresh PUBLIC ExplorerSnapshot shell32 shlwapi comctl32)
add_library(ExplorerDiagnostics STATIC ProbeDiagnostics.cpp)
target_compile_options(ExplorerDiagnostics PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerDiagnostics PUBLIC ExplorerLoadingRefresh oleaut32)
add_library(AssetLibraryExplorerProof SHARED ExplorerProof.cpp NavigationMenu.cpp SnapshotIcon.cpp ExplorerProof.def)
target_compile_definitions(AssetLibraryExplorerProof PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AssetLibraryExplorerProof PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(AssetLibraryExplorerProof PRIVATE ExplorerDiagnostics ExplorerLoadingRefresh ExplorerSnapshot ole32 oleaut32 shell32 user32 propsys shlwapi)
add_executable(ExplorerProofProbe Probe.cpp DirectNavigation.cpp)
target_compile_definitions(ExplorerProofProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExplorerProofProbe PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerProofProbe PRIVATE ole32 oleaut32 shell32 uuid)
add_executable(ExplorerLoaderProbe LoaderProbe.cpp)
# Keep the observer independent of the dynamic CRT imports under investigation.
set_property(TARGET ExplorerLoaderProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(ExplorerLoaderProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExplorerLoaderProbe PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerLoaderProbe PRIVATE ole32 uuid)
add_executable(ExplorerSnapshotProbe SnapshotProbe.cpp)
add_executable(ExplorerSnapshotTests SnapshotTests.cpp)
add_executable(ExplorerNavigationMenuTests NavigationMenuTests.cpp)
foreach(target ExplorerSnapshotProbe ExplorerSnapshotTests ExplorerNavigationMenuTests)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /analyze /utf-8)
  target_link_libraries(${target} PRIVATE ExplorerSnapshot oleaut32 shell32 shlwapi)
endforeach()
enable_testing()
add_test(NAME explorer_snapshot COMMAND ExplorerSnapshotTests $<TARGET_FILE:AssetLibraryExplorerProof> $<TARGET_FILE:ExplorerSnapshotProbe>)
set_tests_properties(explorer_snapshot PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
add_test(NAME explorer_navigation_menu COMMAND ExplorerNavigationMenuTests $<TARGET_FILE:AssetLibraryExplorerProof>)
set_tests_properties(explorer_navigation_menu PROPERTIES TIMEOUT 10)
add_executable(ExplorerLoadingRefreshTests LoadingRefreshTests.cpp)
target_compile_options(ExplorerLoadingRefreshTests PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerLoadingRefreshTests PRIVATE ExplorerLoadingRefresh oleaut32)
add_test(NAME explorer_loading_refresh COMMAND ExplorerLoadingRefreshTests $<TARGET_FILE:AssetLibraryExplorerProof>)
set_tests_properties(explorer_loading_refresh PROPERTIES TIMEOUT 30)
add_test(NAME explorer_loading_view_wiring COMMAND ExplorerLoadingRefreshTests $<TARGET_FILE:AssetLibraryExplorerProof> --wiring-only)
set_tests_properties(explorer_loading_view_wiring PROPERTIES TIMEOUT 10)
add_executable(ExplorerDiagnosticsTests ProbeDiagnosticsTests.cpp)
target_compile_options(ExplorerDiagnosticsTests PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerDiagnosticsTests PRIVATE ExplorerDiagnostics)
add_test(NAME explorer_diagnostics COMMAND ExplorerDiagnosticsTests $<TARGET_FILE:AssetLibraryExplorerProof>)
set_tests_properties(explorer_diagnostics PROPERTIES TIMEOUT 10)
