cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(AssetLibraryExplorerProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(AssetLibraryExplorerProof SHARED ExplorerProof.cpp ExplorerProof.def)
target_compile_definitions(AssetLibraryExplorerProof PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AssetLibraryExplorerProof PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(AssetLibraryExplorerProof PRIVATE ole32 oleaut32 shell32 user32 propsys shlwapi)
add_executable(ExplorerProofProbe Probe.cpp DirectNavigation.cpp)
target_compile_definitions(ExplorerProofProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExplorerProofProbe PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerProofProbe PRIVATE ole32 oleaut32 shell32 uuid)
add_executable(ExplorerLoaderProbe LoaderProbe.cpp)
# Keep the observer independent of the dynamic CRT imports under investigation.
set_property(TARGET ExplorerLoaderProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(ExplorerLoaderProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExplorerLoaderProbe PRIVATE /W4 /WX /permissive- /analyze /utf-8)
target_link_libraries(ExplorerLoaderProbe PRIVATE ole32 uuid)
