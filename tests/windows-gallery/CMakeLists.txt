cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_VERSION "10.0.26100.0")
project(AssetLibraryNativeGalleryHarness LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native gallery harness requires MSVC and Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(surface "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/windows-shell")
add_library(GallerySurface STATIC "${surface}/gallery/Surface.cpp" "${surface}/gallery/Layout.cpp"
  "${surface}/gallery/Accessible.cpp" "${surface}/gallery/Uia.cpp" "${surface}/Snapshot.cpp")
target_include_directories(GallerySurface PUBLIC "${surface}")
target_compile_definitions(GallerySurface PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(GallerySurface PUBLIC user32 gdi32 msimg32 oleacc ole32 oleaut32 comctl32 shell32 advapi32 uuid uiautomationcore)
add_executable(GalleryPreviewTests PreviewTests.cpp)
target_link_libraries(GalleryPreviewTests PRIVATE GallerySurface)
add_executable(GalleryUiaTests UiaTests.cpp)
target_link_libraries(GalleryUiaTests PRIVATE GallerySurface)
add_executable(GalleryLayoutTests LayoutTests.cpp)
target_link_libraries(GalleryLayoutTests PRIVATE GallerySurface)
add_executable(GallerySurfaceTests SurfaceTests.cpp)
target_link_libraries(GallerySurfaceTests PRIVATE GallerySurface)
add_executable(GalleryHarness Harness.cpp)
target_link_libraries(GalleryHarness PRIVATE GallerySurface uiautomationcore)
add_executable(GalleryRenderingTests RenderingTests.cpp)
target_link_libraries(GalleryRenderingTests PRIVATE GallerySurface)
add_executable(GalleryBudgetTests BudgetTests.cpp)
target_link_libraries(GalleryBudgetTests PRIVATE GallerySurface)
foreach(target GalleryPreviewTests GalleryUiaTests GalleryHarness GallerySurfaceTests GalleryRenderingTests GalleryBudgetTests)
  target_link_options(${target} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/Harness.manifest")
endforeach()
add_executable(GalleryAccessibilityProbe AccessibilityProbe.cpp)
target_compile_definitions(GalleryAccessibilityProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(GalleryAccessibilityProbe PRIVATE user32 ole32 oleaut32 oleacc uiautomationcore uuid)
foreach(target GalleryPreviewTests GalleryUiaTests GallerySurface GalleryLayoutTests GallerySurfaceTests GalleryHarness GalleryRenderingTests GalleryBudgetTests GalleryAccessibilityProbe)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /analyze /utf-8)
endforeach()
enable_testing()
add_test(NAME gallery_layout COMMAND GalleryLayoutTests)
add_test(NAME gallery_surface COMMAND GallerySurfaceTests)
add_test(NAME gallery_accessibility_external COMMAND GalleryHarness --external $<TARGET_FILE:GalleryAccessibilityProbe>)
add_test(NAME gallery_accessibility_retired_proxy COMMAND GalleryHarness --external-retire $<TARGET_FILE:GalleryAccessibilityProbe>)
add_test(NAME gallery_rendering COMMAND GalleryRenderingTests)
set_tests_properties(gallery_layout gallery_surface PROPERTIES TIMEOUT 30)
set_tests_properties(gallery_accessibility_external gallery_accessibility_retired_proxy PROPERTIES TIMEOUT 25)
set_tests_properties(gallery_rendering PROPERTIES TIMEOUT 15)

add_test(NAME gallery_budget COMMAND GalleryBudgetTests)
set_tests_properties(gallery_budget PROPERTIES TIMEOUT 15)

add_test(NAME gallery_uia COMMAND GalleryUiaTests)
set_tests_properties(gallery_uia PROPERTIES TIMEOUT 15)

add_test(NAME gallery_accessibility_page COMMAND GalleryHarness --external-page $<TARGET_FILE:GalleryAccessibilityProbe>)
set_tests_properties(gallery_accessibility_page PROPERTIES TIMEOUT 25)

add_test(NAME gallery_accessibility_cycles COMMAND GalleryHarness --retention-cycles $<TARGET_FILE:GalleryAccessibilityProbe>)
set_tests_properties(gallery_accessibility_cycles PROPERTIES TIMEOUT 60)

add_test(NAME gallery_accessibility_listener COMMAND GalleryHarness --listener $<TARGET_FILE:GalleryAccessibilityProbe>)
set_tests_properties(gallery_accessibility_listener PROPERTIES TIMEOUT 25)

add_test(NAME gallery_preview COMMAND GalleryPreviewTests)
set_tests_properties(gallery_preview PROPERTIES TIMEOUT 20)
